  };

  // Print individual record
  const handlePrint = () => {
    window.print();
  };

  const navBtn = (id,lbl,icon) => (

          </button>

          <span className="hide-sm" style={{color:C.muted,fontFamily:"sans-serif",fontSize:"0.8rem"}}>
            <FaUserShield style={{color:C.accent,marginRight:4}}/><strong>{admin.name}</strong>

        {/* ── RECORDS ── */}

          </div>

          {/* Desktop table */}
          <div className="card show-tbl" style={{padding:0,overflow:"hidden"}}>

                            <div style={{display:"flex",alignItems:"center",gap:7}}>
                              <div style={{width:28,height:28,borderRadius:"50%",background:C.light,display:"flex",alignItems:"center",justifyContent:"center",flexShrink:0,color:C.primary,fontSize:"0.75rem",fontWeight:700}}>

                          </td>
                          <td className="tbl-hide-mobile">{row.department}</td>

                          <td className="tbl-hide-mobile" style={{whiteSpace:"nowrap",color:C.muted,fontSize:"0.76rem"}}>{fmtDate(row.dob_day,row.dob_month,row.dob_year)}</td>
                          <td className="tbl-hide-mobile">
                            {row.tveta_reg_no
                              ? <span className="badge-green"><FaCheck/>{row.tveta_reg_no}</span>
                              : <span style={{color:C.muted,fontSize:"0.76rem"}}>—</span>
                            }
                          </td>
                          <td className="tbl-hide-mobile" style={{whiteSpace:"nowrap",color:C.muted,fontSize:"0.73rem"}}>
                            {row.submitted_at ? new Date(row.submitted_at).toLocaleDateString("en-KE",{day:"2-digit",month:"short",year:"numeric"}) : "—"}
                          </td>
                          <td>
                            <div className="tbl-actions">
                              <button className="btn-outline" style={{padding:"4px 9px",fontSize:"0.72rem"}} onClick={()=>setSelected(row)}>
                                <FaEye/> View
                              </button>
                              <button className="btn-danger" style={{padding:"4px 9px",fontSize:"0.72rem"}} onClick={()=>openConfirmModal(row.id,row.full_name)} disabled={deleting===row.id}>

                              </button>
                            </div>
                          </td>
                        </tr>
                      ))}
                    </tbody>
                  </table>
                </div>

                  <div className="rec-card" key={row.id}>
                    <div className="rec-card-header">

                      <div style={{flex:1,minWidth:0}}>
                        <div style={{fontFamily:"sans-serif",fontWeight:700,fontSize:"0.9rem",color:C.primary,overflow:"hidden",textOverflow:"ellipsis",whiteSpace:"nowrap"}}>{row.full_name}</div>

                    </div>
                    <div className="rec-card-footer">
                      <button className="btn-outline" style={{padding:"6px 12px",fontSize:"0.78rem"}} onClick={()=>setSelected(row)}><FaEye/> View</button>

                        {deleting===row.id?<FaSpinner className="spin"/>:<FaTrash/>} Delete
                      </button>
                    </div>
                  </div>

            }
          </div>
        </>}

      {selected && (
        <div className="overlay no-print" onClick={()=>setSelected(null)}>
          <div className="modal" ref={printRef} onClick={e=>e.stopPropagation()}>
            <div className="modal-header">
              <div style={{display:"flex",alignItems:"center",gap:10,flex:1,minWidth:0}}>
                <div style={{width:42,height:42,borderRadius:"50%",background:C.light,display:"flex",alignItems:"center",justifyContent:"center",fontSize:"1rem",fontWeight:700,color:C.primary,flexShrink:0}}>
                  {selected.full_name?.charAt(0)?.toUpperCase()}
                </div>
                <div style={{minWidth:0}}>
                  <h2 style={{fontFamily:"sans-serif",color:C.primary,fontSize:"1rem",marginBottom:4,overflow:"hidden",textOverflow:"ellipsis",whiteSpace:"nowrap"}}>{selected.full_name}</h2>
                  <div style={{display:"flex",gap:5,flexWrap:"wrap"}}>
                    <span className="badge">{selected.designation}</span>
                    <span className="badge">{selected.department}</span>
                    {selected.tveta_reg_no && <span className="badge-green"><FaCertificate/>{selected.tveta_reg_no}</span>}

                </div>
              </div>
              <div className="modal-actions">
                <button className="btn-icon" onClick={handlePrint} title="Print this record"><FaPrint size={16}/></button>

              </div>
            </div>

            {[
              {sec:"Personal Information", icon:<FaUser/>, fields:[
                [<FaUser/>,"Full Name",         selected.full_name],

                [<FaIdCard/>,"National ID",     selected.national_id],
                [<FaCalendarAlt/>,"Date of Birth", fmtDate(selected.dob_day,selected.dob_month,selected.dob_year)],
                [<FaInfoCircle/>,"Disability",  selected.disability],
                [<FaInfoCircle/>,"Disability Nature",selected.disability_nature],
              ]},
              {sec:"Employment Details", icon:<FaBriefcase/>, fields:[
                [<FaUserTie/>,"Designation",    selected.designation],
                [<FaBuilding/>,"Department",    selected.department],
                [<FaBriefcase/>,"Job Group",    selected.job_group],
                [<FaCalendarAlt/>,"Date of Appointment", fmtDate(selected.appointment_day,selected.appointment_month,selected.appointment_year)],
                [<FaCalendarAlt/>,"Retirement/Expiry",   fmtDate(selected.retirement_day,selected.retirement_month,selected.retirement_year)],
                [<FaClipboardList/>,"Duties & Roles",selected.duties_roles],
                [<FaClipboardList/>,"Responsibilities",  selected.responsibilities],
              ]},
              {sec:"Academic Qualifications", icon:<FaGraduationCap/>, fields:[
                [<FaBook/>,"KCSE",              selected.kcse_year?`${selected.kcse_year} — ${selected.kcse_grade}`:""],
                [<FaCertificate/>,"Certificate",selected.cert_name?`${selected.cert_name} (${selected.cert_year}) — ${selected.cert_grade}`:""],
                [<FaCertificate/>,"Diploma",    selected.diploma_name?`${selected.diploma_name} (${selected.diploma_year}) — ${selected.diploma_grade}`:""],
                [<FaCertificate/>,"Higher Diploma",selected.hdip_name?`${selected.hdip_name} (${selected.hdip_year}) — ${selected.hdip_grade}`:""],
                [<FaGraduationCap/>,"Degree",   selected.degree_name?`${selected.degree_name} (${selected.degree_year}) — ${selected.degree_grade}`:""],

                [<FaGraduationCap/>,"PhD",      selected.phd_name?`${selected.phd_name} (${selected.phd_year}) — ${selected.phd_grade}`:""],
              ]},
              {sec:"Professional Development", icon:<FaChalkboardTeacher/>, fields:[
                [<FaChalkboardTeacher/>,"Pedagogy",   selected.pedagogy_year?`${selected.pedagogy_year} — ${selected.pedagogy_grade}`:""],
                [<FaChalkboardTeacher/>,"ToT",        selected.tot_year?`${selected.tot_year} — ${selected.tot_grade}`:""],
                [<FaChalkboardTeacher/>,"Supervisory",selected.superv_year?`${selected.superv_year} — ${selected.superv_grade}`:""],
                [<FaChalkboardTeacher/>,"SLDP",       selected.sldp_year?`${selected.sldp_year} — ${selected.sldp_grade}`:""],
                [<FaChalkboardTeacher/>,"Retirement Course",selected.retire_course_year?`${selected.retire_course_year} — ${selected.retire_course_grade}`:""],
                [<FaCertificate/>,"TVETA Reg No.",    selected.tveta_reg_no],
                [<FaCalendarAlt/>,"TVETA Registered", selected.tveta_date],
                [<FaCalendarAlt/>,"TVETA Expiry",     selected.tveta_expiry],
              ]},
            ].map(({sec,icon,fields})=>{
              const visible=fields.filter(([,,v])=>v);
              if(!visible.length) return null;
              return (
                <div key={sec} style={{marginBottom:"1.1rem"}}>
                  <div className="sec-title">{icon}{sec}</div>
                  {visible.map(([ico,k,v])=>(
                    <div className="rev-row" key={k}>
                      <span className="rev-key">{ico}{k}</span>

                </div>

            })}

            <div style={{display:"flex",gap:8,justifyContent:"flex-end",marginTop:"1.1rem",paddingTop:"1rem",borderTop:`1px solid ${C.border}`,flexWrap:"wrap"}}>
              <button className="btn-outline" onClick={handlePrint}><FaPrint/> Print Record</button>
              <button className="btn-danger" onClick={()=>openConfirmModal(selected.id,selected.full_name)} disabled={deleting===selected.id}>
                {deleting===selected.id?<FaSpinner className="spin"/>:<FaTrash/>} Delete Record
              </button>
              <button className="btn" onClick={()=>setSelected(null)}><FaTimes/> Close</button>
            </div>
          </div>
        </div>
      )}

      {/* ── CONFIRM DELETE MODAL ── */}
      {confirmModal && (
        <div className="overlay" style={{zIndex:200}} onClick={()=>setConfirmModal(null)}>
          <div className="confirm-modal" onClick={e=>e.stopPropagation()}>
            <div style={{width:52,height:52,borderRadius:"50%",background:"#fef2f2",display:"flex",alignItems:"center",justifyContent:"center",margin:"0 auto 1rem",color:C.danger,fontSize:"1.4rem"}}>
              <FaTrash/>
            </div>
            <h3 style={{fontFamily:"sans-serif",color:C.text,marginBottom:8}}>Delete Record?</h3>
            <p style={{fontFamily:"sans-serif",fontSize:"0.83rem",color:C.muted,marginBottom:"1.5rem",lineHeight:1.5}}>

            </p>
            <div style={{display:"flex",gap:8,justifyContent:"center",flexWrap:"wrap"}}>
              <button className="btn-outline" onClick={()=>setConfirmModal(null)}><FaTimes/> Cancel</button>
              <button className="btn-danger" style={{padding:"10px 20px"}} onClick={handleDelete}><FaTrash/> Yes, Delete</button>
            </div>
          </div>
        </div>
      )}
    </div>
  );
}
